
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null &&
        request.auth.uid == 'hmCNDi9OSGRO5gXrzYHEpopB6WD2';
    }

    match /listings/{listingId} {
      allow read: if true;

      allow create: if
        request.auth != null &&
        request.resource.data.keys().hasAll(['title','price','category','desc','img','seller','ownerUid','ts','likes','comments']) &&
        request.resource.data.title is string && request.resource.data.title.size() <= 60 &&
        request.resource.data.price is number && request.resource.data.price >= 0 &&
        request.resource.data.desc is string && request.resource.data.desc.size() <= 500 &&
        request.resource.data.img is string &&
        request.resource.data.seller is string && request.resource.data.seller.size() <= 20 &&
        request.resource.data.ownerUid == request.auth.uid &&
        request.resource.data.likes is list &&
        request.resource.data.comments is list;

      allow update: if request.auth != null &&
        (isAdmin() || resource.data.ownerUid == request.auth.uid ||
          (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likes', 'comments']) &&
           request.resource.data.likes is list &&
           request.resource.data.comments is list &&
           request.resource.data.ownerUid == resource.data.ownerUid));
      allow delete: if request.auth != null &&
        (isAdmin() || resource.data.ownerUid == request.auth.uid);
    }

    match /conversations/{conversationId} {
      allow read: if request.auth != null &&
        request.auth.uid in resource.data.participants;
      allow create: if request.auth != null &&
        request.auth.uid in request.resource.data.participants &&
        request.resource.data.participants is list &&
        request.resource.data.participants.size() == 2 &&
        request.resource.data.sellerUid is string &&
        request.resource.data.buyerUid is string &&
        request.resource.data.listingId is string &&
        request.resource.data.listingTitle is string;
      allow update: if request.auth != null &&
        request.auth.uid in resource.data.participants &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['lastMessage', 'updatedAt']);

      match /messages/{messageId} {
        allow read: if request.auth != null &&
          request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants;
        allow create: if request.auth != null &&
          request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participants &&
          request.resource.data.senderUid == request.auth.uid &&
          request.resource.data.sender is string &&
          request.resource.data.text is string &&
          request.resource.data.text.size() <= 500 &&
          request.resource.data.ts is number;
      }
    }
  }
}
